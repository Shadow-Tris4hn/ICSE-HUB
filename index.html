Connect to cloud silently in the background bindStudentProfile(savedId); } } async function handleFirstTimeLogin(e) { e.preventDefault(); const name = document.getElementById('firstTimeNameInput').value.trim(); const lang = document.getElementById('firstTimeLangChoice').value; const app = document.getElementById('firstTimeAppChoice').value; if (!name) return; // Stable persistent device-based ID let studentId = localStorage.getItem("current_icse_student_id"); }); // Save locally first so refresh never forgets you localStorage.setItem("current_icse_student_id", studentId); localStorage.setItem("current_student_profile_cache", JSON.stringify(newStudentDoc)); // Sync to Firebase if (typeof db !== "undefined") { try { await db.collection("students").doc(studentId).set(newStudentDoc, { merge: true }); } catch(err) { console.error("Cloud save error:", err); } } document.getElementById('firstTimeLoginModal').classList.add('hidden'); bindStudentProfile(studentId); } function bindStudentProfile(studentId) { if (typeof db === "undefined") return; db.collection("students").doc(studentId).onSnapshot((snap) => { if (snap.exists) { currentStudent = snap.data(); // Update cache with freshest cloud copy localStorage.setItem("current_student_profile_cache", JSON.stringify(currentStudent)); document.getElementById('navStudentName').innerText = currentStudent.name; document.getElementById('studentSubjectChoicesSub').innerText = `Electives: ${currentStudent.langChoice || 'Bengali'} & ${currentStudent.appChoice || 'Computer Applications'}`; initSubjectDropdowns(); renderSubjectsView(); computeAnalyticsAndPercentiles(); renderStudentTickets(); } else if (currentStudent) { // If cloud document is missing but local cache exists, push local profile back up db.collection("students").doc(studentId).set(currentStudent, { merge: true }); } }, (err) => { console.warn("Firestore offline or busy, running on local persistent cache:", err); }); }
